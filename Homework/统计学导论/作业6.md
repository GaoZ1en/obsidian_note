### 问题1

```
MC_int <- function(f,a,b,n){
  random_samples <- runif(n,a,b)
  function_values <- sapply(random_samples, f)
  MC <- (b-a)*mean(function_values)
  return(MC)
}

f <- function(x){return(x^2*exp(x^2))}
a <- 0
b <- 1

result <- MC_int(f,a,b,100000)

print(paste("结果为:", result))

rm(list=ls())
```

这里定义了用于计算
$$
\int^b_af(x){\rm d}x
$$
的可调节精确度的MC模拟程序。将$f=x^2e^{x^2},a=0,b=1$代入，并取$n=100000$，运行结果为：

### 问题3

```
mu <- 2
sigma <- 3
n <- 30   
N <- 1000

Z <- replicate(N, {
  X <- rnorm(n, mu, sigma)
  mean_X <- mean(X)
  (sqrt(n) * (mean_X - mu)) / sigma
})

F_1000 <- function(x) {sum(Z <= x) / N}

x_val <- seq(-3, 3, by = 0.5)

empi_cdf_val <- sapply(x_val, F_1000)

theo_cdf_val <- pnorm(x_val)

comp <- data.frame(
  x = x_val,
  Empi = empi_cdf_val,
  Theo = theo_cdf_val
)

print(comp)

rm(list = ls())
```
