```R
# 模拟实验的次数
num_simulations <- 10000

# 存储样本方差和统计量T的结果
sample_variances <- numeric(num_simulations)
T_values <- numeric(num_simulations)

# 进行模拟
for (i in 1:num_simulations) {
  # 从均匀分布U(0,1)中生成样本
  sample <- runif(3, min = 0, max = 1)
  
  # 计算样本均值
  sample_mean <- mean(sample)
  
  # 计算样本方差 S^2
  sample_variance <- var(sample)
  
  # 计算统计量 T
  T_value <- (1/3) * sum((sample - sample_mean)^2)
  
  # 存储结果
  sample_variances[i] <- sample_variance
  T_values[i] <- T_value
}

# 计算两个估计值的均值和方差
mean_sample_variance <- mean(sample_variances)
mean_T_value <- mean(T_values)

var_sample_variance <- var(sample_variances)
var_T_value <- var(T_values)

# 打印结果
cat("样本方差的平均值: ", mean_sample_variance, "\n")
cat("统计量 T 的平均值: ", mean_T_value, "\n")
cat("样本方差的方差: ", var_sample_variance, "\n")
cat("统计量 T 的方差: ", var_T_value, "\n")

# 删除对象，释放内存
rm(list = ls())
gc()
```

极大似然估计法步骤如下：

假设有一组按照概率密度$\displaystyle{f(\theta_{1},\theta_{2},\dots,\theta_{k}|x)}$且独立同分布的样本$\displaystyle{X_{1},X_{2},\dots,X_{n}}$，构造似然函数

$$
\begin{align}
L(\theta_{1},\theta_{2},\dots,\theta_{k}|X)=\prod^{n}_{i=1}f(\theta_{1},\theta_{2},\dots,\theta_{k}|X_{i})
\end{align}
$$

然后有

$$
\begin{align}
\log L(\theta|X)=\ell(\theta|X)=\sum^{n}_{i=1}\ln f(\theta|X_{i})
\end{align}
$$

求导

$$
\begin{align}
\sum^{n}_{i=1} \frac{f'_{\theta_{i}}(\theta|X_{i})}{f(\theta|X_{i})}=0\implies \theta_{i}=\theta_{i}^{*}
\end{align}
$$

极大似然估计法是无偏估计，且在所有的无偏估计中，方差最小；随着样本量的增加，估计值收敛于真实值。
