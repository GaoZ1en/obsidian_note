\xi _{\pm1} & =\frac{1}{2}e^{\pm(\tau-i\phi)}\left( ir(1+r^{2})^{-1/2} \partial _{\tau}\mp i(1+r^{2})^{1/2} \partial _{r}-r^{-1}(1+r^{2})^{1/2}\partial _{\phi} \right) \\

\end{align}$$
