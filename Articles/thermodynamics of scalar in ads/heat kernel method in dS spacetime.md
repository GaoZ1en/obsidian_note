$$\begin{align}
K(t,\rho) & =\frac{1}{2\pi ^{2}}\sum ^{\infty}_{n=0} (n+1) \frac{\sin(n+1)\rho}{\sin \rho} e^{-n(n+2)t}
\end{align}$$
