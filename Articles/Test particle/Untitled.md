\begin{align}
0 =\delta _{\xi}S_{p} & =\int \mathrm{d}^{4}x\sqrt{ -g }\nabla _{\mu}[T^{\mu \nu}\xi _{\nu}+j^{\mu}A^{\nu}\xi _{\nu}]-\int \mathrm{d}\tau  \frac{\mathrm{d}}{\mathrm{d}\tau}\left((mU_{\nu}+qA_{\nu})\xi ^{\nu}\right)+(\text{ bulk terms })

$$\tag{1.3}
\begin{align}
T^{\mu \nu} & = m\int \mathrm{d}\tau U^{\mu}U^{\nu} \frac{\delta ^{4}(x-x(\tau))}{\sqrt{ -g }} \\

\end{align}
$$
