the gauge fixed eom is given by

$$\begin{align}
\nabla ^{(0)2}h_{\mu \nu}+2h_{\mu \nu} & =0
\end{align}$$

in addition, we impose the TT gauge, i.e.,

$$\begin{align}
h & =g^{\mu \nu}h_{\mu \nu}=0 \\
\nabla ^{(0)} _{\mu}h^{\mu \nu} & =0
\end{align}$$

expand these equations in the coordinate $\displaystyle{(t,r,\phi)}$ with metric

$$\begin{align}
\mathrm{d}s^{2} & =-f(r)\mathrm{d}t^{2}+\frac{\mathrm{d}r^{2}}{f(r)}+r^{2}\mathrm{d}\phi ^{2}
\end{align}$$

with $f(r)=1+r^{2}$,

we have

$$\begin{align}
0={}&\mathcal{L}h_{tt}-\frac{2}{f}h_{tt}-2r^{2}f h_{rr}-4r\partial_{r}h_{tt}+4r\partial_{t}h_{tr},\\
0={}&\mathcal{L}h_{tr}-\frac{3r^{4}+1}{r^{2}f}h_{tr}+2r\partial_{t}h_{rr}+\frac{2r}{f^{2}}\partial_{t}h_{tt}-\frac{2}{r^{3}}\partial_{\phi}h_{t\phi},\\
0={}&\mathcal{L}h_{t\phi}+2r\partial_{t}h_{r\phi}-\frac{2(2r^{2}+1)}{r}\partial_{r}h_{t\phi}+\frac{2f}{r}\partial_{\phi}h_{tr},\\
0={}&\mathcal{L}h_{rr}+4r\partial_{r}h_{rr}+\frac{2(3r^{4}+r^{2}-1)}{r^{2}f}h_{rr}+\frac{4r}{f^{2}}\partial_{t}h_{tr}-\frac{2r^{2}}{f^{3}}h_{tt}+\frac{2}{r^{4}}h_{\phi\phi}-\frac{4}{r^{3}}\partial_{\phi}h_{r\phi},\\
0={}&\mathcal{L}h_{r\phi}-\frac{2}{r}\partial_{r}h_{r\phi}-\frac{3f}{r^{2}}h_{r\phi}+\frac{2f}{r}\partial_{\phi}h_{rr}+\frac{2r}{f^{2}}\partial_{t}h_{t\phi}-\frac{2}{r^{3}}\partial_{\phi}h_{\phi\phi},\\
0={}&\mathcal{L}h_{\phi\phi}+\frac{2}{r^{2}}h_{\phi\phi}+\frac{4f}{r}\left(\partial_{\phi}h_{r\phi}-\partial_{r}h_{\phi\phi}\right)+2f^{2}h_{rr}.
\end{align}$$

here

$$\begin{align}
\mathcal{L}X&=f\partial_{r}^{2}X+\frac{1+3r^{2}}{r}\partial_{r}X-\frac{1}{f}\partial_{t}^{2}X+\frac{1}{r^{2}}\partial_{\phi}^{2}X 
\end{align}$$

and

$$\begin{align}
-\frac{h_{tt}}{f}+f h_{rr}+\frac{h_{\phi\phi}}{r^{2}}&=0 .
\end{align}$$

$$\begin{align}
-\frac{1}{f}\partial _{t}h_{tt}+f\partial _{r}h_{tr}+\frac{1+3r^{2}}{r}h_{tr}+\frac{1}{r^{2}}\partial _{\phi}h_{t\phi} & =0 \\
-\frac{1}{f}\partial _{t}h_{tr}+f\partial _{r}h_{rr}+\frac{1+4r^{2}}{r}h_{rr}+\frac{1}{r^{2}}\partial _{\phi}h_{r\phi}-\frac{1}{r^{3}}h_{\phi \phi}+\frac{r}{f^{2}}h_{tt} & =0 \\
-\frac{1}{f}\partial _{t}h_{t\phi}+f\partial _{r}h_{r\phi}+\frac{1+3r^{2}}{r}h_{r\phi}+\frac{1}{r^{2}}\partial _{\phi}h_{\phi \phi} & =0
\end{align}$$

---

now fix the residual gauge redundancy. a convenient choice is to assume the onshell configuration is a pure gauge configuration. that is, generated by a vector $\displaystyle{\zeta ^{\mu}}$ as

$$\begin{align}
h_{\mu \nu} & =\nabla ^{(0)} _{\mu}\zeta _{\nu}+\nabla ^{(0)} _{\nu}\zeta _{\mu}.
\end{align}$$

Writing the lower-index components as$\zeta_{\mu}=(\zeta_{t},\zeta_{r},\zeta_{\phi})$, this gives

$$\begin{align}
h_{tt}&=2\partial_{t}\zeta_{t}-2rf\zeta_{r},\\
h_{tr}&=\partial_{t}\zeta_{r}+\partial_{r}\zeta_{t}-\frac{2r}{f}\zeta_{t},\\
h_{t\phi}&=\partial_{t}\zeta_{\phi}+\partial_{\phi}\zeta_{t},\\
h_{rr}&=2\partial_{r}\zeta_{r}+\frac{2r}{f}\zeta_{r},\\
h_{r\phi}&=\partial_{r}\zeta_{\phi}+\partial_{\phi}\zeta_{r}-\frac{2}{r}\zeta_{\phi},\\
h_{\phi\phi}&=2\partial_{\phi}\zeta_{\phi}+2rf\zeta_{r}.
\end{align}$$

Substituting this into the TT gauge condition gives

$$\begin{align}
h & =2\nabla ^{(0)}_{\mu}\zeta ^{\mu}=0 \\
\nabla ^{(0)\mu}h_{\mu \nu} & =(\nabla ^{(0)2}-2)\zeta _{\nu}+\nabla ^{(0)}_{\nu}\nabla ^{(0)}_{\mu}\zeta ^{\mu}=0 \\
\implies \nabla ^{(0)}_{\mu}\zeta ^{\mu} & =0 \\
(\nabla ^{(0)2}-2)\zeta ^{\mu} & =0
\end{align}$$

naïve solutions are

$$\begin{align}
\zeta ^{\mu}_{(2,1)} & \propto \frac{e^{-4it}}{(1+r^{2})^{2}}\bar{\xi} _{1}^{\mu} \\
\zeta ^{\mu}_{(1,2)} & \propto \frac{e^{-4it}}{(1+r^{2})^{2}}\xi_{1}^{\mu}
\end{align}$$

which form the $\displaystyle{(2,1)}$ and $\displaystyle{(1,2)}$ reps. of the isometry group $\displaystyle{\mathrm{SL}(2,\mathbb{R})\times \mathrm{SL}(2,\mathbb{R})}$. but notice that

$$\begin{align}
\zeta ^{\mu}_{(2,1)} & \propto \mathcal{L}_{\bar{\xi}_{-1}} \zeta ^{\mu}_{L} \\
\zeta ^{\mu}_{(1,2)} & \propto \mathcal{L}_{\xi _{-1}}\zeta _{R}^{\mu}
\end{align}$$

and the Lie derivatives commute with $\displaystyle{\nabla ^{(0)}}$, we have

$$\begin{align}
h_{L} & =\mathcal{L}_{\zeta _{L}}g^{(0)}, & h_{R} & =\mathcal{L}_{\zeta _{R}}g^{(0)}
\end{align}$$

are also solutions of linearised Einstein equation.

---

the second order operator in the gauge fixed linearized eom

$$\begin{align}
(\nabla ^{(0)2}+2)h_{\mu \nu} & =0
\end{align}$$

can be decomposed into two first-order operators

$$\begin{align}
(\varepsilon _{\mu}^{~\nu \rho}\nabla ^{(0)}_{\nu}+\delta ^{\rho}_{\mu})(\varepsilon _{\rho}^{~\sigma \lambda}\nabla _{\sigma} ^{(0)}-\delta _{\rho}^{\lambda})h_{\lambda \kappa} & =0


\end{align}$$

therefore the solutions to the eom can be rewritten as

$$\begin{align}
h_{\mu \nu} & =h^{(+)}_{\mu \nu}+h^{(-)}_{\mu \nu}
\end{align}$$

with

$$\begin{align}
(\varepsilon _{\mu}^{~\nu \rho}\nabla ^{(0)}_{\nu}\pm \delta _{\mu}^{\rho})h^{(\pm)}_{\rho \sigma} & =0
\end{align}$$

now we will solve two first order equations alone. first consider

$$\begin{align}
\varepsilon _{\mu}^{~\nu \rho}\nabla ^{(0)}_{\nu}h_{\rho \sigma}+h_{\mu \sigma} & =0
\end{align}$$

we expand

$$\begin{align}
h_{\mu \nu}(x) & =h_{ab}(x) \xi _{a,\mu}(x)\xi _{b,\nu}(x), \quad a,b\in\left\{-1,0,1\right\}
\end{align}$$

remembering that 

$$\begin{align}
\nabla _{\mu}\xi _{a,\nu} & =-\varepsilon _{\mu \nu \rho}\xi _{a}^{\rho} \\
\nabla _{\mu}\bar{\xi}_{a,\nu} & =\varepsilon _{\mu \nu \rho}\bar{\xi}_{a}^{\rho}
\end{align}$$

then the gauge fixing conditions give

$$\begin{align}
 & h_{ab} =h_{ba} \\
h=0\implies  & h_{1,-1}-\frac{1}{4}h_{00}=0, \\
\nabla ^{(0)}_{\mu}h^{\mu \nu}=0\implies & \mathcal{L}_{\xi _{1}}h_{11}+\mathcal{L}_{\xi _{0}}h_{10}+\mathcal{L}_{\xi _{-1}}h_{1,-1}=0, \\
 & \mathcal{L}_{\xi _{1}}h_{10}+\mathcal{L}_{\xi _{0}}h_{00}+\mathcal{L}_{\xi _{-1}}h_{0,-1}=0, \\
 & \mathcal{L}_{\xi _{1}}h_{1,-1}+\mathcal{L}_{\xi _{0}}h_{0,-1}+\mathcal{L}_{\xi _{-1}}h_{-1,-1}=0
\end{align}$$

and the first order equation $\displaystyle{\varepsilon _{\mu}^{~\nu \rho}\nabla ^{(0)}_{\nu}h_{\rho \sigma}+h_{\mu \sigma}=0}$ gives

$$\begin{align}
E_{11}&\equiv (2i\mathcal{L}_{\xi _{0}}-4)h_{11}+i\mathcal{L}_{\xi _{-1}}h_{10}=0, \\
E_{10}&\equiv (2i\mathcal{L}_{\xi _{0}}-4)h_{10}-i\mathcal{L}_{\xi _{-1}}h_{00}-2i\mathcal{L}_{\xi _{1}}h_{11}=0, \\
E_{1,-1}&\equiv (2i\mathcal{L}_{\xi _{0}}-2)h_{1,-1}-i\mathcal{L}_{\xi _{-1}}h_{0,-1}-\frac{1}{2}h_{00}=0, \\
E_{0,1}&\equiv 2i\mathcal{L}_{\xi _{-1}}h_{1,-1}-4h_{10}-2i\mathcal{L}_{\xi _{1}}h_{11}=0, \\
E_{00}&\equiv -2i\mathcal{L}_{\xi _{1}}h_{10}-4h_{1,-1}-3h_{00}+2i\mathcal{L}_{\xi _{-1}}h_{0,-1}=0, \\
E_{0,-1}&\equiv -2i\mathcal{L}_{\xi _{1}}h_{1,-1}-4h_{0,-1}-2i\mathcal{L}_{\xi _{-1}}h_{-1,-1}=0, \\
E_{-1,1}&\equiv -i\mathcal{L}_{\xi _{1}}h_{10}-(2+2i\mathcal{L}_{\xi _{0}})h_{1,-1}-\frac{1}{2}h_{00}=0, \\
E_{-1,0}&\equiv -i\mathcal{L}_{\xi _{1}}h_{00}-(4+2i\mathcal{L}_{\xi _{0}})h_{0,-1}=0, \\
E_{-1,-1}&\equiv -i\mathcal{L}_{\xi _{1}}h_{0,-1}-(4+2i\mathcal{L}_{\xi _{0}})h_{-1,-1}=0
\end{align}$$

then we can solve out the coefficients $\displaystyle{h_{ab}(x)}$ as

$$\begin{align}
h_{10}&=h_{1,-1}=h_{00}=h_{0,-1}=h_{-1,-1}=0, \\
\mathcal{L}_{\xi _{1}}h_{11}&=0, \\
\mathcal{L}_{\xi _{0}}h_{11}&=-2ih_{11}.
\end{align}$$

therefore we writing

$$\begin{align}
w & \equiv \frac{r\,e^{-i(t+\phi)}}{\sqrt{ 1+r^{2} }}.
\end{align}$$

we have $\displaystyle{\mathcal{L}_{\xi _{1}}w=\mathcal{L}_{\xi _{0}}w=0}$, so the general solution is

$$\begin{align}
h_{11}(t,r,\phi) & =\frac{e^{-4it}}{(1+r^{2})^{2}}G(w),
\end{align}$$

with $\displaystyle{G}$ arbitrary. Imposing single-valuedness in $\displaystyle{\phi}$ and regularity at $\displaystyle{r=0}$ gives the power-series expansion

$$\begin{align}
G(w) & =\sum _{n=0}^{\infty}\tilde{c}_{n}w^{n},
\end{align}$$

so that

$$\begin{align}
h_{11}(t,r,\phi) & =\sum _{n=0}^{\infty}\tilde{c}_{n}\frac{r^{n}e^{-i(n+4)t-in\phi}}{(1+r^{2})^{2+n/2}}.
\end{align}$$

similarly for

$$\begin{align}
\varepsilon _{\mu}^{~\nu \rho}\nabla ^{(0)}_{\nu}h_{\rho \sigma}-h_{\mu \sigma}=0
\end{align}$$

we expand

$$\begin{align}
h_{\mu \nu} & =h_{ab}\bar{\xi} _{a,\mu}\bar{\xi} _{b,\nu}
\end{align}$$

and then

$$\begin{align}
 & h_{ab}=h_{ba}, \\
h=0\implies & h_{1,-1}-\frac{1}{4}h_{00}=0, \\
\nabla ^{(0)}_{\mu}h^{\mu \nu}=0\implies & \mathcal{L}_{\bar{\xi}_{1}}h_{11}+\mathcal{L}_{\bar{\xi}_{0}}h_{10}+\mathcal{L}_{\bar{\xi}_{-1}}h_{1,-1}=0, \\
 & \mathcal{L}_{\bar{\xi}_{1}}h_{10}+\mathcal{L}_{\bar{\xi}_{0}}h_{00}+\mathcal{L}_{\bar{\xi}_{-1}}h_{0,-1}=0, \\
 & \mathcal{L}_{\bar{\xi}_{1}}h_{1,-1}+\mathcal{L}_{\bar{\xi}_{0}}h_{0,-1}+\mathcal{L}_{\bar{\xi}_{-1}}h_{-1,-1}=0
\end{align}$$

and the first-order equation gives

$$\begin{align}
\tilde{E}_{11}&\equiv (4-2i\mathcal{L}_{\bar{\xi}_{0}})h_{11}-i\mathcal{L}_{\bar{\xi}_{-1}}h_{10}=0, \\
\tilde{E}_{10}&\equiv (4-2i\mathcal{L}_{\bar{\xi}_{0}})h_{10}-i\mathcal{L}_{\bar{\xi}_{-1}}h_{00}+2i\mathcal{L}_{\bar{\xi}_{1}}h_{11}=0, \\
\tilde{E}_{1,-1}&\equiv (2-2i\mathcal{L}_{\bar{\xi}_{0}})h_{1,-1}-i\mathcal{L}_{\bar{\xi}_{-1}}h_{0,-1}+\frac{1}{2}h_{00}=0, \\
\tilde{E}_{0,1}&\equiv 2i\mathcal{L}_{\bar{\xi}_{1}}h_{11}+4h_{10}-2i\mathcal{L}_{\bar{\xi}_{-1}}h_{1,-1}=0, \\
\tilde{E}_{00}&\equiv 2i\mathcal{L}_{\bar{\xi}_{1}}h_{10}+4h_{1,-1}+3h_{00}-2i\mathcal{L}_{\bar{\xi}_{-1}}h_{0,-1}=0, \\
\tilde{E}_{0,-1}&\equiv 2i\mathcal{L}_{\bar{\xi}_{1}}h_{1,-1}+4h_{0,-1}-2i\mathcal{L}_{\bar{\xi}_{-1}}h_{-1,-1}=0, \\
\tilde{E}_{-1,1}&\equiv i\mathcal{L}_{\bar{\xi}_{1}}h_{10}+(2+2i\mathcal{L}_{\bar{\xi}_{0}})h_{1,-1}+\frac{1}{2}h_{00}=0, \\
\tilde{E}_{-1,0}&\equiv i\mathcal{L}_{\bar{\xi}_{1}}h_{00}+(4+2i\mathcal{L}_{\bar{\xi}_{0}})h_{0,-1}=0, \\
\tilde{E}_{-1,-1}&\equiv i\mathcal{L}_{\bar{\xi}_{1}}h_{0,-1}+(4+2i\mathcal{L}_{\bar{\xi}_{0}})h_{-1,-1}=0
\end{align}$$

Solving these equations together with the TT constraints gives

$$\begin{align}
h_{10}&=h_{1,-1}=h_{00}=h_{0,-1}=h_{-1,-1}=0, \\
\mathcal{L}_{\bar{\xi}_{1}}h_{11}&=0, \\
\mathcal{L}_{\bar{\xi}_{0}}h_{11}&=-2ih_{11}.
\end{align}$$

write

$$\begin{align}
z & \equiv \frac{r\,e^{-i(t-\phi)}}{\sqrt{ 1+r^{2} }},
\end{align}$$

Since $\displaystyle{\mathcal{L}_{\bar{\xi}_{1}}z=\mathcal{L}_{\bar{\xi}_{0}}z=0}$, the general solution is

$$\begin{align}
h_{11}(t,r,\phi) & =\frac{e^{-4it}}{(1+r^{2})^{2}}F(z),
\end{align}$$

with $\displaystyle{F}$ arbitrary. Imposing single-valuedness in $\displaystyle{\phi}$ and regularity at $\displaystyle{r=0}$ gives

$$\begin{align}
F(z) & =\sum _{n=0}^{\infty}c_{n}z^{n},
\end{align}$$

so that

$$\begin{align}
h_{11}(t,r,\phi) & =\sum _{n=0}^{\infty}c_{n}\frac{r^{n}e^{-i(n+4)t+in\phi}}{(1+r^{2})^{2+n/2}}.
\end{align}$$

non-physical modes are obtained by taking $\displaystyle{\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}}}$ of $\displaystyle{h_{11}}$.
