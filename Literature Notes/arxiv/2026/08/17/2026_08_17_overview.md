  \[
  \{\Omega(x_1),\mu(x_2)\}=\widetilde\delta^{(3)}(x_1,x_2),
  \qquad
  \{q_{ab}(x_1),q_{cd}(x_2)\}=2P_{abcd}(x_1,x_2).
