   datS, counterexample, ps, cs, ks, pairSum, hModel,

  eps3[a_, b_, c_] := Signature[{a, b, c}];
  eps4[a_, b_, c_, d_] := Signature[{a, b, c, d}];

  dlt[a_, b_] := KroneckerDelta[a, b];
