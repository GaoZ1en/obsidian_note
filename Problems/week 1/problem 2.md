# problem 2

use highest weight representation, find the explicit form (up to a overall constant) of all the solution of equation of motion for free massive vector field with mass $\displaystyle{\mu}$ defined in global $\displaystyle{\mathrm{AdS}_{3}}$. here we choose the following metric

$$\begin{align}
\mathrm{d}s^{2} & =-(1+r^{2})\mathrm{d}t^{2}+\frac{\mathrm{d}r^{2}}{1+r^{2}}+r^{2}\mathrm{d}\phi ^{2}
\end{align}$$

the independent Killing vectors are

$$\begin{align}
\xi _{1} & =\frac{1}{2}e^{i(t-\phi)}\left( \frac{r}{\sqrt{ 1+r^{2} }}\partial _{t}-i\sqrt{ 1+r^{2} }\partial _{r}-\frac{\sqrt{ 1+r^{2} }}{r}\partial _{\phi} \right) \\
\xi _{0} & =\frac{1}{2}(\partial _{t}-\partial _{\phi}) \\
\xi _{-1} & =\frac{1}{2}e^{-i(t-\phi)}\left( \frac{r}{\sqrt{ 1+r^{2} }}\partial _{t}+i\sqrt{ 1+r^{2} }\partial _{r}-\frac{\sqrt{ 1+r^{2} }}{r}\partial _{\phi} \right) \\
\bar{\xi}_{1} & =\frac{1}{2}e^{i(t+\phi)}\left( \frac{r}{\sqrt{ 1+r^{2} }}\partial _{t}-i\sqrt{ 1+r^{2} }\partial _{r}+\frac{\sqrt{ 1+r^{2} }}{r}\partial _{\phi} \right) \\
\bar{\xi}_{0} & =\frac{1}{2}(\partial _{t}+\partial _{\phi}) \\
\bar{\xi}_{-1} & =\frac{1}{2}e^{-i(t+\phi)}\left(\frac{r}{\sqrt{ 1+r^{2} }}\partial _{t}+i\sqrt{ 1+r^{2} }\partial _{r}+\frac{\sqrt{ 1+r^{2} }}{r}\partial _{\phi}\right)
\end{align}$$

which satisfies the $\displaystyle{\mathfrak{sl}(2,\mathbb{R})\times \mathfrak{sl}(2,\mathbb{R})}$ algebra

$$\begin{align}
[\xi _{0},\xi _{1}] & =i\xi _{1} \\
[\xi _{0},\xi _{-1}] & =-i\xi _{-1} \\
[\xi _{1},\xi _{-1}] & =-2i\xi _{0} \\
[\bar{\xi} _{0},\bar{\xi} _{1}] & =i\bar{\xi} _{1} \\
[\bar{\xi} _{0},\bar{\xi} _{-1}] & =-i\bar{\xi} _{-1} \\
[\bar{\xi} _{1},\bar{\xi} _{-1}] & =-2i\bar{\xi} _{0}
\end{align}$$

## solution

the Casimir operators can be constructed as

$$\begin{align}
\mathcal{C} & =\mathcal{L}_{\xi_{0}}\mathcal{L}_{\xi_{0}}-\mathcal{L}_{\xi _{-1}}\mathcal{L}_{\xi _{1}}+i\mathcal{L}_{\xi_{0}} \\
\mathcal{\bar{C}} & =\mathcal{L}_{\bar{\xi}_{0}}\mathcal{L}_{\bar{\xi}_{0}}-\mathcal{L}_{\bar{\xi}_{-1}}\mathcal{L}_{\bar{\xi} _{1}}+i\mathcal{L}_{\bar{\xi}_{0}}
\end{align}$$

which acts on vector field as

$$\begin{align}
(\mathcal{C}+\mathcal{\bar{C}})A^{\mu} & =-\frac{1}{2}\nabla ^{2}A^{\mu}-A^{\mu} \\
(\mathcal{C}-\mathcal{\bar{C}})^{2}A^{\mu} & =\nabla ^{2}A^{\mu}-\nabla _{\nu}\nabla ^{\mu}A^{\nu}
\end{align}$$

notice the eom for massive vector field can be written as

$$\begin{align}
\nabla _{\mu}F^{\mu \nu} & =\mu ^{2}A^{\nu} \\
\implies & \begin{cases}
\nabla _{\mu}A^{\mu} =0 \\
\nabla ^{2}A^{\mu}-(\mu ^{2}-2)A^{\mu}=0
\end{cases}
\end{align}$$

then the Casimirs acting on $A^{\mu}$ give

$$\begin{align}
(\mathcal{C}+\mathcal{\bar{C}})A^{\mu} & =-\frac{1}{2}\mu ^{2}A^{\mu} \\
(\mathcal{C}-\mathcal{\bar{C}})^{2}A^{\mu}  & =\mu ^{2}A^{\mu}
\end{align}$$

which gives the two eigenvalue of the Casimirs, corresponding to two series of highest weight representations (positive frequency modes):

$$\begin{align}
\begin{cases}
\mathcal{C}=-\frac{1}{4}\mu ^{2}-\frac{1}{2}\mu \\
\mathcal{\bar{C}}=-\frac{1}{4}\mu ^{2}+\frac{1}{2}\mu
\end{cases} \\
\text{or }\begin{cases}
\mathcal{C}=-\frac{1}{4}\mu ^{2}+\frac{1}{2}\mu \\
\mathcal{\bar{C}}=-\frac{1}{4}\mu ^{2}-\frac{1}{2}\mu
\end{cases}
\end{align}$$

for the first case, we can define the highest weight state $\displaystyle{f^{\mu}_{l,0,0}}$ as

$$\begin{align}
\mathcal{L}_{\xi _{1}}{f^{\mu}_{l,0,0}} & =0 \\
\mathcal{L}_{\bar{\xi} _{1}}{f^{\mu}_{l,0,0}} & =0
\end{align}$$

notice that

$$\begin{align}
\mathcal{L}_{\xi_{1}}\xi _{1}^{\mu}=\mathcal{L}_{\bar{\xi}_{1}}\xi_{1}^{\mu}=0
\end{align}$$

thus the highest weight state can be solved as

$$\begin{align}
f_{l,0,0} & =C_{0,0}e^{-i(\mu+2)t}(1+r^{2})^{-(\mu+2)/2}\xi _{1}^{\mu}
\end{align}$$

where $\displaystyle{C_{0,0}}$ is an overall constant. the other states can be generated by acting the lowering operators

$$\begin{align}
f^{\mu}_{l,n,\bar{n}} & =C_{n,\bar{n}}\mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi} _{-1}}^{\bar{n}}f^{\mu}_{l,0,0}
\end{align}$$

by induction, we can find

$$\begin{align}
f^{\mu}_{l,n,\bar{n}} & = \mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}}\xi _{1}^{\mu}+2in\mathcal{L}^{n-1}_{\xi _{-1}}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}}-n(n-1)\mathcal{L}_{\xi _{-1}}^{n-2}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}}\xi _{-1}^{\mu} \\
\mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}} & =C_{n,\bar{n}}e^{-i(\mu+2+n+\bar{n})t}e^{i(n-\bar{n})\phi}r^{n-\bar{n}}(1+r^{2})^{-(\mu+2+n-\bar{n})/2}P_{\bar{n}}^{(\mu+1,n-\bar{n})}\left(\frac{r^{2}-1}{r^{2}+1}\right)
\end{align}$$

as for the second case, we can define the highest weight state $\displaystyle{f^{\mu}_{r,0,0}}$ as

$$\begin{align}
\mathcal{L}_{\xi _{1}}{f^{\mu}_{r,0,0}} & =0 \\
\mathcal{L}_{\bar{\xi} _{1}}{f^{\mu}_{r,0,0}} & =0
\end{align}$$


notice that

$$\begin{align}
\mathcal{L}_{\xi_{1}}\bar{\xi} _{1}^{\mu}=\mathcal{L}_{\bar{\xi}_{1}}\bar{\xi}_{1}^{\mu}=0
\end{align}$$

thus the highest weight state can be solved as

$$\begin{align}
f_{r,0,0} & =\tilde{C}_{0,0}e^{-i(\mu+2)t}(1+r^{2})^{-(\mu+2)/2}\bar{\xi} _{1}^{\mu}
\end{align}$$

where $\displaystyle{\tilde{C}_{0,0}}$ is an overall constant. the other states can be generated by acting the lowering operators

$$\begin{align}
f^{\mu}_{r,n,\bar{n}} & =\tilde{C}_{n,\bar{n}}\mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi} _{-1}}^{\bar{n}}f^{\mu}_{r,0,0}
\end{align}$$

by induction, we can find

$$\begin{align}
f^{\mu}_{r,n,\bar{n}} & = \mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}}\bar{\xi} _{1}^{\mu}+2i\bar{n}\mathcal{L}^{n}_{\xi _{-1}}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}-1}-\bar{n}(\bar{n}-1)\mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}-2} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}}\bar{\xi} _{-1}^{\mu} \\
\mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}} & =C_{n,\bar{n}}e^{-i(\mu+2+n+\bar{n})t}e^{i(n-\bar{n})\phi}r^{n-\bar{n}}(1+r^{2})^{-(\mu+2+n-\bar{n})/2}P_{\bar{n}}^{(\mu+1,n-\bar{n})}\left(\frac{r^{2}-1}{r^{2}+1}\right)
\end{align}$$

as a summary, the explicit form of all the solutions of equation of motion for free massive vector field with mass $\displaystyle{m}$ defined in global $\displaystyle{\mathrm{AdS}_{3}}$ are given by

$$\begin{align}
f^{\mu}_{l,n,\bar{n}} & = \mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}}\xi _{1}^{\mu}+2in\mathcal{L}^{n-1}_{\xi _{-1}}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}}-n(n-1)\mathcal{L}_{\xi _{-1}}^{n-2}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}}\xi _{-1}^{\mu} \\ \\
f^{\mu}_{r,n,\bar{n}} & = \mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}}\bar{\xi} _{1}^{\mu}+2i\bar{n}\mathcal{L}^{n}_{\xi _{-1}}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}-1}-\bar{n}(\bar{n}-1)\mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}-2} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}}\bar{\xi} _{-1}^{\mu} \\
\mathcal{L}_{\xi _{-1}}^{n}\mathcal{L}_{\bar{\xi}_{-1}}^{\bar{n}} \frac{e^{-i(\mu+2)t}}{(1+r^{2})^{(\mu +2)/2}} & =C_{n,\bar{n}}e^{-i(\mu+2+n+\bar{n})t}e^{i(n-\bar{n})\phi}r^{n-\bar{n}}(1+r^{2})^{-(\mu+2+n-\bar{n})/2}P_{\bar{n}}^{(\mu+1,n-\bar{n})}\left(\frac{r^{2}-1}{r^{2}+1}\right)
\end{align}$$
